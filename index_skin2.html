/* ========================================================= Velvet Room — Skin2 Inventory Filter (SAFE) File: assets/js/inventorySkin2Filter.js Fix: - NO MutationObserver (can cause infinite loops) - Filters only on user input or when data is rendered Depends on: - #filter input - #status span - table rows: - inventorySkin2Page.js dispatches: "skin2:inventoryRendered" ========================================================= */ const elFilter = document.getElementById("filter"); const elStatus = document.getElementById("status"); function setStatus(text) { if (elStatus) elStatus.textContent = text; } function getRows() { return Array.from(document.querySelectorAll("tr.inv-row")); } function applyFilter() { const queryRaw = elFilter ? elFilter.value : ""; const q = (queryRaw || "").trim().toLowerCase(); const rows = getRows(); const total = rows.length; // If table isn't rendered yet, do nothing. if (!total) return; let visible = 0; if (!q) { for (const r of rows) { r.style.display = ""; visible++; } setStatus(`Loaded ${total} rows`); return; } for (const r of rows) { const hay = (r.getAttribute("data-search") || "").toLowerCase(); const show = hay.includes(q); r.style.display = show ? "" : "none"; if (show) visible++; } setStatus(`Showing ${visible} of ${total} (filter: "${queryRaw}")`); } function debounce(fn, waitMs = 80) { let t = null; return () => { if (t) clearTimeout(t); t = setTimeout(fn, waitMs); }; } const onInput = debounce(applyFilter); // Wire: input filtering if (elFilter) { elFilter.addEventListener("input", onInput); } // Wire: apply once data is rendered (from inventorySkin2Page.js) window.addEventListener("skin2:inventoryRendered", () => { applyFilter(); });
