import { supabase } from "./supabaseClient.js"; import { rotatingStarSVG } from "./ui/star.js"; const VIEW_NAME = "v_coterie"; const ROW_LIMIT = 300; const elContent = document.getElementById("coterie-content"); const elStatus = document.getElementById("status"); const elError = document.getElementById("error"); function setStatus(text) { if (elStatus) elStatus.textContent = text ?? ""; } function escapeHtml(v) { return String(v ?? "") .replaceAll("&", "&") .replaceAll("<", "<") .replaceAll(">", ">"); } function renderTable(rows) { if (!rows?.length) { elContent.innerHTML = "No Coterie rows returned."; return; } const tbody = rows.map(r => { const star = r.new_update ? rotatingStarSVG({ size: 16, style: "margin-right:6px;" }) : ""; const bottleLink = ` ${escapeHtml(r.bottle_expression)} `; const addLink = `
Add your sensory notes to the Coterie
`; return ` ${star}${bottleLink}${addLink} ${escapeHtml(r.score)} ${escapeHtml(r.msrp)} ${escapeHtml(r.proof)} ${escapeHtml(r.age)} `; }).join(""); elContent.innerHTML = ` ${tbody}
Bottle Score MSRP Proof Age
`; window.dispatchEvent(new Event("skin2:inventoryRendered")); } async function load() { setStatus("Loading coterie…"); const { data, error } = await supabase.from(VIEW_NAME).select("*").limit(ROW_LIMIT); if (error) { elError.textContent = error.message; return; } renderTable(data || []); setStatus(`Loaded ${data?.length ?? 0} rows`); } load();
